cmake_minimum_required(VERSION 3.25)
project(SegaRallyNativeVulkanModel2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Vulkan 1.1 REQUIRED)
set(COMMON "${ROOT}/tests/test_native_vulkan_model2.cpp" "${ROOT}/src/app/gpu/model2_frame.cpp"
    "${ROOT}/src/runtime/video.cpp" "${ROOT}/src/runtime/raster.cpp")
add_library(segarally_native_vulkan STATIC "${ROOT}/src/app/native/vulkan/model2_vulkan.cpp" "${ROOT}/src/app/gpu/model2_frame.cpp")
target_include_directories(segarally_native_vulkan PUBLIC "${ROOT}/src")
target_link_libraries(segarally_native_vulkan PUBLIC Vulkan::Vulkan)
add_executable(test_native_vulkan_model2 "${ROOT}/tests/test_native_vulkan_model2.cpp" "${ROOT}/src/runtime/video.cpp" "${ROOT}/src/runtime/raster.cpp")
target_include_directories(test_native_vulkan_model2 PRIVATE "${ROOT}/src")
target_compile_definitions(test_native_vulkan_model2 PRIVATE NATIVE_VULKAN_MODEL2=1)
target_link_libraries(test_native_vulkan_model2 PRIVATE segarally_native_vulkan)
add_executable(test_native_vulkan_async "${ROOT}/tests/test_native_vulkan_async.cpp" "${ROOT}/src/runtime/video.cpp" "${ROOT}/src/runtime/raster.cpp")
target_link_libraries(test_native_vulkan_async PRIVATE segarally_native_vulkan)
foreach(target segarally_native_vulkan test_native_vulkan_model2 test_native_vulkan_async)
if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /fp:strict /utf-8)
else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers
        -Wno-misleading-indentation -fno-fast-math -ffp-contract=off)
endif()
endforeach()
option(SEGARALLY_VULKAN_PRESENTATION "Build direct Vulkan SDL window/surface fixture" OFF)
option(SEGARALLY_VULKAN_SDL_REFERENCE "Build separate pinned SDL Vulkan comparison binary, not linked by native target" OFF)
if(SEGARALLY_VULKAN_SDL_REFERENCE OR SEGARALLY_VULKAN_PRESENTATION)
    set(SEGARALLY_SDL_SOURCE "${ROOT}/extern/sdl3" CACHE PATH "Pinned SDL reference source")
    execute_process(COMMAND git -C "${SEGARALLY_SDL_SOURCE}" rev-parse HEAD
        OUTPUT_VARIABLE REVISION OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT REVISION STREQUAL "fa2c02bb6e21974a89ea9824bc53c9932abe5f9c")
        message(FATAL_ERROR "SDL reference must be the pinned revision")
    endif()
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    if(UNIX AND NOT APPLE)
        set(SDL_X11 ${SEGARALLY_VULKAN_PRESENTATION} CACHE BOOL "" FORCE)
        set(SDL_WAYLAND OFF CACHE BOOL "" FORCE)
        set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
    endif()
    if(NOT SEGARALLY_VULKAN_SDL_REFERENCE)
        set(SDL_GPU OFF CACHE BOOL "" FORCE)
    endif()
    add_subdirectory("${SEGARALLY_SDL_SOURCE}" "${CMAKE_BINARY_DIR}/sdl3" EXCLUDE_FROM_ALL)
    if(SEGARALLY_VULKAN_SDL_REFERENCE)
    add_executable(test_sdl_vulkan_model2 ${COMMON} "${ROOT}/src/app/gpu/gpu_renderer.cpp")
    target_include_directories(test_sdl_vulkan_model2 PRIVATE "${ROOT}/src")
    target_link_libraries(test_sdl_vulkan_model2 PRIVATE SDL3::SDL3-static)
    if(MSVC)
        target_compile_options(test_sdl_vulkan_model2 PRIVATE /fp:strict /utf-8)
    else()
        target_compile_options(test_sdl_vulkan_model2 PRIVATE -fno-fast-math -ffp-contract=off)
    endif()
    endif()
    if(SEGARALLY_VULKAN_PRESENTATION)
        add_executable(test_native_vulkan_presenter "${ROOT}/tests/test_native_vulkan_presenter.cpp"
            "${ROOT}/src/app/native/vulkan/vulkan_presenter.cpp" "${ROOT}/src/runtime/video.cpp" "${ROOT}/src/runtime/raster.cpp")
        target_link_libraries(test_native_vulkan_presenter PRIVATE segarally_native_vulkan SDL3::SDL3-static)
        if(UNIX AND NOT APPLE)
            find_package(X11 REQUIRED)
            target_link_libraries(test_native_vulkan_presenter PRIVATE X11::X11)
        endif()
        if(MSVC)
            target_compile_options(test_native_vulkan_presenter PRIVATE /W4 /fp:strict /utf-8)
        else()
            target_compile_options(test_native_vulkan_presenter PRIVATE -Wall -Wextra -Werror
                -Wno-missing-field-initializers -Wno-misleading-indentation -fno-fast-math -ffp-contract=off)
        endif()
    endif()
endif()
