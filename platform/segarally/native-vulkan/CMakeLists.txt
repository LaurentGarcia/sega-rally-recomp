cmake_minimum_required(VERSION 3.25)
project(SegaRallyNativeVulkan LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan 1.1 REQUIRED)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(test_segarally_native_vulkan crt_smoke.cpp)
target_include_directories(test_segarally_native_vulkan PRIVATE "${ROOT}/src")
target_link_libraries(test_segarally_native_vulkan PRIVATE Vulkan::Vulkan)
if(MSVC)
    target_compile_options(test_segarally_native_vulkan PRIVATE /W4 /fp:strict /utf-8)
else()
    target_compile_options(test_segarally_native_vulkan PRIVATE -Wall -Wextra -Werror -Wno-missing-field-initializers -fno-fast-math -ffp-contract=off)
endif()
enable_testing()
add_test(NAME segarally_native_vulkan_crt COMMAND test_segarally_native_vulkan)
set_tests_properties(segarally_native_vulkan_crt PROPERTIES TIMEOUT 30)
